# Configure this standalone project after `flutter build windows` to opt in.
cmake_minimum_required(VERSION 3.14)
project(flex_switch_windows_utils_tests LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Windows utility tests require the Windows SDK and runtime.")
endif()

set(FLUTTER_EPHEMERAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../flutter/ephemeral"
  CACHE PATH "Flutter Windows headers and libraries generated by flutter build")
if(NOT EXISTS "${FLUTTER_EPHEMERAL_DIR}/flutter_windows.dll.lib")
  message(FATAL_ERROR "Run flutter build windows before configuring native tests.")
endif()

enable_testing()
add_executable(flex_switch_utils_test
  "utils_test.cpp"
  "../runner/utils.cpp"
)
target_compile_features(flex_switch_utils_test PRIVATE cxx_std_17)
target_compile_options(flex_switch_utils_test PRIVATE /W4 /WX /EHsc)
target_compile_definitions(flex_switch_utils_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(flex_switch_utils_test PRIVATE
  "../runner"
  "${FLUTTER_EPHEMERAL_DIR}"
)
target_link_libraries(flex_switch_utils_test PRIVATE
  "${FLUTTER_EPHEMERAL_DIR}/flutter_windows.dll.lib"
  "shell32.lib"
)
add_custom_command(TARGET flex_switch_utils_test POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${FLUTTER_EPHEMERAL_DIR}/flutter_windows.dll"
    "$<TARGET_FILE_DIR:flex_switch_utils_test>"
  VERBATIM
)
add_test(NAME utf8_from_utf16 COMMAND flex_switch_utils_test)
set_tests_properties(utf8_from_utf16 PROPERTIES TIMEOUT 30)
